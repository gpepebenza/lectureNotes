def diff(big: Z, small: Z): Z = {
    Contract(
        Requires( big >= small ),
        Ensures ( Res[Z] == big-small )
    )

    //PREMISES:
    big >= small


    var count: Z = 0
    var cur: Z = big

    //PREMISES:
    big >= small (precondition can always list as a premise)
    count == 0
    cur == big

    //SHOW:
    count == (big-cur) (must prove all invariants before loop)


    while (cur != small) {
        Invariant(
            Modifies(count, cur),
            count == (big-cur)
        )

        //PREMISES:
        big >= small
        count == (big-cur) (list all invariants as premise at beginning of each iteration)
        cur != small (loop condition is true)

        cur = cur - 1
        count = count + 1

        //SHOW:
        count == (big-cur) (must prove all invariants still hold at end of iteration)

    }

    //PREMISES:
    big >= small
    count == (big-cur) (can list all invariants as premises after loop ends)
    cur == small


    //SHOW:
    count == big-small (prove all postconditions)


    return count
}
